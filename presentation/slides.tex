	\begin{frame}{Defense Approach 1: Refresh-Centric}
		\textbf{Refreshing the Victims}
		\begin{itemize}
			\item \textbf{How it works:} Restore charge before it leaks
			\item \textbf{The Tradeoff:} Either: Keeping track of which rows need refreshing $\rightarrow$ complex tracking arrays.
            
            Or: Probabilistically refresh rows.
			\item \textbf{The Flaw:} $T_{RH}$ falls $\rightarrow$ more refresh
		\end{itemize}
        \begin{figure}[h]
            \centering
            \includegraphics[width=0.7\columnwidth]{refresh.png}
        \end{figure}
	\end{frame}
	
	\begin{frame}{Defense Approach 2: Frequency-Centric}
		\textbf{Stopping the Attacker}
		\begin{itemize}
			\item \textbf{How it works:} Set a speed limit.
			\item \textbf{The Flaw:} False alarms cost too much.
            
            Lower $T_{RH} \rightarrow$ More false alarms  
		\end{itemize}
        \begin{figure}[h]
            \centering
            \includegraphics[width=\columnwidth]{frequency.png}
        \end{figure}
	\end{frame}
	
	\begin{frame}{Defense Approach 3: Isolation-Centric}
		\textbf{Building Walls and Moving Houses}
		\begin{itemize}
			\item \textbf{How it works:} Breaks the physical closeness of the attacker and victim. 
			\item e.g., by placing empty "guard rows" in between
			\item \textbf{The Trade-off:} Space
			\begin{figure}[h]
				\raggedright
				\includegraphics[width=\columnwidth]{isolation.png}
			\end{figure}
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Blindspots: We Weren't Looking at the Whole Picture}
		\textbf{The Rules Are Changing}
		\begin{itemize}
			\item \textbf{RowPress:} A newer attack variant showed that bits can flip just by keeping a row \textit{open} for a long time
			\item Breaks the fundamental assumption
			\item \textbf{Timing Side-Channels:} Smart attackers can observe how long the defense mechanisms take to activate and use that timing data to steal secrets invisibly
		\end{itemize}
	\end{frame}

	\begin{frame}{Looking Forward}
		\begin{itemize}
			\item We need hybrid approaches.
